1. START HADOOP DAEMONS & VERIFY CLUSTER STATUS
--------------------------------------------------------------------------------

start-all.cmd

# Verify active Java processes

# Check HDFS cluster health and report
hdfs dfsadmin -report


2. INSTALL PYTHON DEPENDENCIES
--------------------------------------------------------------------------------
pip install matplotlib pandas openpyxl pillow


3. HDFS DIRECTORY SETUP & DATA INGESTION
--------------------------------------------------------------------------------
# Create directory structure in HDFS
hdfs dfs -mkdir -p /taxi_project/input/raw
hdfs dfs -mkdir -p /taxi_project/input/cleaned
hdfs dfs -mkdir -p /taxi_project/output

# Upload cleaned CSV dataset to HDFS
hdfs dfs -put -f cleaned_yellow_tripdata.csv /taxi_project/input/cleaned/

# Verify file upload
hdfs dfs -ls -h /taxi_project/input/cleaned/


4. EXECUTE MAPREDUCE STREAMING JOBS (HADOOP STREAMING)
--------------------------------------------------------------------------------
# A. Hourly Demand Analysis
hdfs dfs -rm -r -f /taxi_project/output/hourly
hadoop jar %HADOOP_HOME%\share\hadoop\tools\lib\hadoop-streaming-*.jar ^
  -files mappers\mapper_hourly.py,reducers\reducer_hourly.py ^
  -input /taxi_project/input/cleaned ^
  -output /taxi_project/output/hourly ^
  -mapper "python mapper_hourly.py" ^
  -reducer "python reducer_hourly.py"

9. CLEANUP & SUBMISSION PACKAGING
--------------------------------------------------------------------------------
# Remove Python cache directories
rmdir /s /q __pycache__ 2>nul
for /d /r %i in (__pycache__) do @rmdir /s /q "%i" 2>nul

# Package full submission ZIP archive
powershell -Command "Compress-Archive -Path * -DestinationPath ..\hdfs_project_submission.zip -Force"

# Package lightweight ZIP archive (excluding raw CSV)
powershell -Command "Get-ChildItem -Path . -Exclude 'cleaned_yellow_tripdata.csv', '*.parquet', '*.tar.gz' | Compress-Archive -DestinationPath ..\hdfs_project_submission.zip -Force"
